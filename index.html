# تابعی که تعداد اعضای مجموعه را محاسبه می‌کند def count_elements(s): return len(s) # دریافت ورودی مجموعه به صورت رشته input_set = input("مجموعه را وارد کنید (مثال: {1, 2, 3, 4}): ") # تبدیل ورودی به مجموعه input_set = eval(input_set) # نمایش تعداد اعضای مجموعه print("تعداد اعضای مجموعه: ", count_elements(input_set))
